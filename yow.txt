<script>

function updateJam(){
var now=new Date();
var j=String(now.getHours()).padStart(2,'0');
var m=String(now.getMinutes()).padStart(2,'0');
var s=String(now.getSeconds()).padStart(2,'0');
document.getElementById('jam').textContent=j+':'+m+':'+s;
var hari=['Minggu','Senin','Selasa','Rabu','Kamis','Jumat','Sabtu'];
var bulan=['Jan','Feb','Mar','Apr','Mei','Jun','Jul','Agu','Sep','Okt','Nov','Des'];
document.getElementById('tanggal').textContent=hari[now.getDay()]+', '+now.getDate()+' '+bulan[now.getMonth()]+' '+now.getFullYear();
}

updateJam();
setInterval(updateJam,1000);

var mulaiWaktu=0;
var timer=null;
var berjalan=false;
var elapsed=0;

function tampilkanSw(ms){
var total=Math.floor(ms/1000);
var menit=Math.floor(total/60);
var detik=total%60;
var milli=Math.floor((ms%1000)/10);
document.getElementById('menit').textContent=String(menit).padStart(2,'0');
document.getElementById('detik').textContent=String(detik).padStart(2,'0');
document.getElementById('ms').textContent=String(milli).padStart(2,'0');
}

function swMulai(){
if(berjalan)return;
berjalan=true;
mulaiWaktu=Date.now()-elapsed;
timer=setInterval(function(){
elapsed=Date.now()-mulaiWaktu;
tampilkanSw(elapsed);
},10);
document.getElementById('btnStart').disabled=true;
document.getElementById('btnStop').disabled=false;
}

function swStop(){
if(!berjalan)return;
berjalan=false;
clearInterval(timer);
document.getElementById('btnStart').disabled=false;
document.getElementById('btnStop').disabled=true;
}

function swReset(){
swStop();
elapsed=0;
tampilkanSw(0);
}

document.addEventListener('keydown',function(e){
if(e.code==='Space'){
e.preventDefault();
if(berjalan)swStop();
else swMulai();
}
});
